cmake_minimum_required(VERSION 3.20)
project(crispmesh VERSION 0.1.0 LANGUAGES CXX)
include(cmake/OpenCV.cmake)

option(CRISPMESH_BUILD_TESTS "Build native tests" ON)

add_library(crispmesh_core STATIC core/src/core.cpp)
target_include_directories(crispmesh_core PUBLIC core/include)
target_compile_features(crispmesh_core PUBLIC cxx_std_20)
set_target_properties(crispmesh_core PROPERTIES CXX_EXTENSIONS OFF)
if(CRISPMESH_WITH_OPENCV)
  target_link_libraries(crispmesh_core PUBLIC ${CRISPMESH_OPENCV_LIBRARIES})
  target_include_directories(crispmesh_core PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
  target_compile_definitions(crispmesh_core PUBLIC CRISPMESH_WITH_OPENCV=1)
endif()

add_executable(crispmesh core/src/cli.cpp)
target_link_libraries(crispmesh PRIVATE crispmesh_core)

if(CRISPMESH_BUILD_TESTS)
  enable_testing()
  add_executable(crispmesh_tests core/tests/core_tests.cpp)
  target_link_libraries(crispmesh_tests PRIVATE crispmesh_core)
  add_test(NAME core_contract COMMAND crispmesh_tests)
  add_test(NAME cli_capabilities COMMAND crispmesh capabilities)
  add_test(NAME cli_geometry COMMAND crispmesh diagnose-geometry)
  add_test(NAME cli_unavailable COMMAND crispmesh reconstruct ${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts/empty-project.json)
  set_tests_properties(cli_unavailable PROPERTIES WILL_FAIL TRUE)
  if(CRISPMESH_WITH_OPENCV)
    add_subdirectory(core/depth)
    add_library(crispmesh_stereo_eval_lib STATIC core/evaluation/stereo_eval.cpp)
    target_include_directories(crispmesh_stereo_eval_lib PUBLIC core/evaluation PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
    target_link_libraries(crispmesh_stereo_eval_lib PUBLIC ${CRISPMESH_OPENCV_LIBRARIES})
    target_compile_features(crispmesh_stereo_eval_lib PUBLIC cxx_std_20)
    add_executable(crispmesh_stereo_eval core/evaluation/main.cpp)
    target_link_libraries(crispmesh_stereo_eval PRIVATE crispmesh_stereo_eval_lib)
    target_include_directories(crispmesh_stereo_eval PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
    add_executable(crispmesh_stereo_eval_tests core/evaluation/tests.cpp)
    target_link_libraries(crispmesh_stereo_eval_tests PRIVATE crispmesh_stereo_eval_lib)
    target_include_directories(crispmesh_stereo_eval_tests PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
    add_test(NAME stereo_eval_synthetic COMMAND crispmesh_stereo_eval_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME stereo_eval_prepared COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/core/evaluation/prepared_smoke.py
        $<TARGET_FILE:crispmesh_stereo_eval> ${CMAKE_CURRENT_SOURCE_DIR}/.local-tools/tmp)
    endif()
    add_executable(crispmesh_pose_integration core/tests/pose_integration.cpp)
    target_link_libraries(crispmesh_pose_integration PRIVATE crispmesh_core)
    target_include_directories(crispmesh_pose_integration PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
    add_test(NAME pose_synthetic_image COMMAND crispmesh_pose_integration
      ${CMAKE_CURRENT_BINARY_DIR}/synthetic-pose-fixture $<TARGET_FILE:crispmesh>)
    add_executable(crispmesh_sparse_integration core/tests/sparse_integration.cpp)
    target_link_libraries(crispmesh_sparse_integration PRIVATE crispmesh_core)
    target_include_directories(crispmesh_sparse_integration PRIVATE ${CRISPMESH_OPENCV_INCLUDE_DIRS})
    add_test(NAME sparse_synthetic_image COMMAND crispmesh_sparse_integration
      ${CMAKE_CURRENT_BINARY_DIR}/synthetic-sparse-fixture $<TARGET_FILE:crispmesh>)
  endif()
endif()
