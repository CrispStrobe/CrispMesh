# This directory is experimental and is added only by the OpenCV test build.
add_library(crispmesh_depth_geometry STATIC depth_geometry.cpp)
target_include_directories(crispmesh_depth_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CRISPMESH_OPENCV_INCLUDE_DIRS})
target_link_libraries(crispmesh_depth_geometry PUBLIC ${CRISPMESH_OPENCV_LIBRARIES})
target_compile_features(crispmesh_depth_geometry PUBLIC cxx_std_20)

add_executable(crispmesh_depth_geometry_tests tests.cpp)
target_link_libraries(crispmesh_depth_geometry_tests PRIVATE crispmesh_depth_geometry)
add_test(NAME depth_geometry_synthetic COMMAND crispmesh_depth_geometry_tests)

add_executable(crispmesh_depth_image_tests image_tests.cpp)
target_link_libraries(crispmesh_depth_image_tests PRIVATE crispmesh_depth_geometry)
add_test(NAME depth_image_synthetic COMMAND crispmesh_depth_image_tests ${CMAKE_BINARY_DIR}/depth-image-fixtures)

add_executable(crispmesh_depth_image_fusion_tests image_fusion_tests.cpp)
target_link_libraries(crispmesh_depth_image_fusion_tests PRIVATE crispmesh_depth_geometry)
add_test(NAME depth_image_fusion_synthetic COMMAND crispmesh_depth_image_fusion_tests ${CMAKE_BINARY_DIR}/depth-image-fusion-fixtures)
